// Fallback working songs if JS file fails to load const fallbackSongs = [ { id: "vS3_72h9-jg", title: "Tere Bina Zindagi Se", artist: "Lata & Kishore" }, { id: "h0L8iB8b-08", title: "Ek Ajnabee Haseena Se", artist: "Kishore Kumar" }, { id: "1V_394V8E1c", title: "Pal Pal Dil Ke Paas", artist: "Kishore Kumar" } ]; let currentIndex = 0; let ytPlayer = null; let isPlaying = false; function getSongs() { if (window.sadabaharSongs && window.sadabaharSongs.length > 0) return window.sadabaharSongs; if (window.sadhabharSongs && window.sadhabharSongs.length > 0) return window.sadhabharSongs; return fallbackSongs; } function updateUI() { const list = getSongs(); if (list.length > 0) { const song = list[currentIndex]; document.getElementById('song-title').innerText = song.title; document.getElementById('artist-name').innerText = song.artist; } } function onYouTubeIframeAPIReady() { ytPlayer = new YT.Player('yt-player', { height: '100', width: '100', playerVars: { 'autoplay': 1, 'controls': 0, 'playsinline': 1 }, events: { 'onStateChange': function(event) { if (event.data === YT.PlayerState.PLAYING) { isPlaying = true; document.getElementById('play-btn').innerText = '❚❚'; document.getElementById('cd-art').classList.add('playing'); } else if (event.data === YT.PlayerState.PAUSED) { isPlaying = false; document.getElementById('play-btn').innerText = '▶'; document.getElementById('cd-art').classList.remove('playing'); } else if (event.data === YT.PlayerState.ENDED) { isPlaying = false; document.getElementById('play-btn').innerText = '▶'; document.getElementById('cd-art').classList.remove('playing'); // गाना खत्म होने पर खुद अगला गाना चलाए nextSong(); } }, // 🔥 अगर गाना ब्लॉक/खराब हो तो तुरंत अगला गाना चलाओ 'onError': function(event) { console.log("YouTube Error Code:", event.data); document.getElementById('song-title').innerText = "Skipping restricted song..."; setTimeout(() => { nextSong(); }, 1200); } } }); } function playCurrentSong() { const list = getSongs(); const song = list[currentIndex]; if (ytPlayer && ytPlayer.loadVideoById) { ytPlayer.loadVideoById(song.id); ytPlayer.playVideo(); } } function togglePlay() { if (!ytPlayer || typeof ytPlayer.playVideo !== 'function') { return alert("Player loading, please try again in 2 seconds."); } if (isPlaying) { ytPlayer.pauseVideo(); } else { if (ytPlayer.getPlayerState && ytPlayer.getPlayerState() === YT.PlayerState.PAUSED) { ytPlayer.playVideo(); } else { playCurrentSong(); } } } function nextSong() { const list = getSongs(); currentIndex = (currentIndex + 1) % list.length; updateUI(); playCurrentSong(); } function prevSong() { const list = getSongs(); currentIndex = (currentIndex - 1 + list.length) % list.length; updateUI(); playCurrentSong(); } function seekAudio(val) { if (ytPlayer && typeof ytPlayer.getDuration === 'function') { const dur = ytPlayer.getDuration(); if (dur > 0) ytPlayer.seekTo((val / 100) * dur, true); } } setInterval(() => { if (ytPlayer && isPlaying && typeof ytPlayer.getDuration === 'function') { const dur = ytPlayer.getDuration(); const cur = ytPlayer.getCurrentTime(); if (dur > 0) { document.getElementById('seekbar').value = (cur / dur) * 100; } } }, 500); window.onload = function() { updateUI(); };
